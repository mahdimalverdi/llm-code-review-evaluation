\definecolor{taxblue}{RGB}{38, 87, 139}
\definecolor{taxlight}{RGB}{238, 244, 251}
\definecolor{taxline}{RGB}{70, 70, 70}

\resizebox{0.94\linewidth}{!}{%
\begin{tikzpicture}[
  font=\sffamily,
  >=Latex,
  question/.style={
    draw=taxblue,
    rounded corners=4pt,
    line width=0.7pt,
    fill=taxlight,
    align=center,
    text width=4.1cm,
    minimum height=0.82cm,
    font=\sffamily\small\bfseries,
  },
  answer/.style={
    draw=taxblue,
    rounded corners=3pt,
    line width=0.55pt,
    fill=white,
    align=center,
    text width=3.85cm,
    minimum height=0.72cm,
    font=\sffamily\scriptsize,
  },
  arrow/.style={->, line width=0.55pt, draw=taxline},
]

\node[question] (evidence) at (0,0) {Is the claim supported by available evidence?};
\node[answer, below left=0.65cm and 0.25cm of evidence] (unsupported) {No evidence but confident claim\\$\rightarrow$ Unsupported};
\node[answer, below right=0.65cm and 0.25cm of evidence] (context) {Missing evidence is central\\$\rightarrow$ Context-dependent};

\node[question, right=1.2cm of evidence] (truth) {Does visible evidence show the claim is false?};
\node[answer, below=0.65cm of truth] (incorrect) {False current behavior or API claim\\$\rightarrow$ Incorrect technical claim};

\node[question, right=1.2cm of truth] (fix) {Is the concern valid but the proposed fix unsafe?};
\node[answer, below=0.65cm of fix] (invalid) {Unsafe or behavior-changing fix\\$\rightarrow$ Invalid fix + recoverable signal};

\node[question, below=2.25cm of truth] (action) {Can the useful signal be preserved?};
\node[answer, below left=0.65cm and 0.35cm of action] (nonaction) {Too vague to preserve\\$\rightarrow$ Non-actionable};
\node[answer, below right=0.65cm and 0.35cm of action] (recover) {Concern and next step identifiable\\$\rightarrow$ Recoverable signal};

\draw[arrow] (evidence) -- (unsupported);
\draw[arrow] (evidence) -- (context);
\draw[arrow] (truth) -- (incorrect);
\draw[arrow] (fix) -- (invalid);
\draw[arrow] (action) -- (nonaction);
\draw[arrow] (action) -- (recover);

\node[below=0.45cm of action, font=\sffamily\footnotesize\itshape, text=taxline, yshift=-1.6cm]
  {Boundary rules prioritize evidence, falsifiability, recoverability, and actionability.};

\end{tikzpicture}%
}
